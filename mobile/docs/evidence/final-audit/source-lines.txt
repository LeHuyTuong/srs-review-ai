  88: /// Danh sách tài liệu theo dự án với bốn phạm vi và thứ tự ưu tiên xử lý từ React.
  89: class ProjectDocumentsScreen extends ConsumerWidget {
  90:   const ProjectDocumentsScreen({required this.projectId, super.key});
  91:   final String projectId;
  92:   @override
  93:   Widget build(BuildContext context, WidgetRef ref) => AcademicPage(
  94:     path: '/projects/$projectId/documents',
  95:     builder: (s) {
  96:       final p = s.project(projectId);
  97:       if (p == null) {
  98:         return [const AppEmptyState(title: 'Không tìm thấy dự án')];
  99:       }
 100:       final docs = s.projectDocuments(p);
 101:       final key = 'documents:$projectId';
 102:       final tab = ref.watch(managementListProvider(key)).tab;
 103:       final vm = ref.read(managementListProvider(key).notifier);
 104:       bool match(AcademicRecord d, int tab) => switch (tab) {
 105:         1 => ['waiting', 'inReview'].contains(d.text('supervisorStatus')),
 106:         2 => d.text('supervisorStatus') == 'needsRevision',
 107:         3 => d.flag('submissionLocked'),
 108:         _ => true,
 109:       };
 110:       int priority(AcademicRecord d) =>
 111:           const {'inReview': 0, 'waiting': 1, 'needsRevision': 2}[d.text(
 112:             'supervisorStatus',
 113:           )] ??
 114:           3;
 115:       final visible = docs.where((d) => match(d, tab)).toList()
 116:         ..sort((a, b) {
 117:           final cmp = priority(a).compareTo(priority(b));
 118:           return cmp != 0

 159: /// Snapshot liên kết học vụ và lịch sử chỉ đọc, dùng chung giữa sáu route phase 2.

 162:   int get revision => integer('revision');
 163:   String get currentSemesterId => text('currentSemesterId');
 164:   List<Course> get courses =>
 165:       values('courses')
 166:           .map((v) => Course(v as AcademicJson))
 167:           .toList(growable: false);
 168:   List<AcademicSemester> get semesters =>
 169:       values('semesters')
 170:           .map((v) => AcademicSemester(v as AcademicJson))
 171:           .toList(growable: false);
 172:   List<CourseOffering> get offerings =>
 173:       values('offerings')
 174:           .map((v) => CourseOffering(v as AcademicJson))
 175:           .toList(growable: false);
 176:   List<ClassRoom> get classes =>
 177:       values('classes')
 178:           .map((v) => ClassRoom(v as AcademicJson))
 179:           .toList(growable: false);
 180:   List<RuleSet> get rules =>
 181:       values('ruleSets')
 182:           .map((v) => RuleSet(v as AcademicJson))
 183:           .toList(growable: false);
 184:   List<StudentGroup> get groups =>
 185:       values('groups')
 186:           .map((v) => StudentGroup(v as AcademicJson))

 189:       values('students')
 190:           .map((v) => AcademicStudent(v as AcademicJson))
 191:           .toList(growable: false);
 192:   List<AcademicRecord> records(String key) =>
 193:       values(key)
 194:           .map((v) => AcademicRecord(v as AcademicJson))
 195:           .toList(growable: false);
 196:   Course? course(String id) => courses.where((v) => v.id == id).firstOrNull;
 197:   CourseOffering? offering(String id) =>
 198:       offerings.where((v) => v.id == id).firstOrNull;
 199:   ClassRoom? classRoom(String id) =>
 200:       classes.where((v) => v.id == id).firstOrNull;
 201:   RuleSet? rule(String id) => rules.where((v) => v.id == id).firstOrNull;
 202:   AcademicSemester? semester(String id) =>
 203:       semesters.where((v) => v.id == id).firstOrNull;
 204:   List<CourseOffering> courseOfferings(String id) =>
 205:       offerings.where((v) => v.courseId == id).toList();
 206:   List<ClassRoom> offeringClasses(CourseOffering off) => classes
 207:       .where(
 208:         (v) =>
 209:             v.offeringId == off.id ||
 210:             (v.semesterId == off.semesterId && v.courseId == off.courseId),
 211:       )
 212:       .toList();
 213:   List<StudentGroup> classGroups(String id) =>
 214:       groups.where((v) => v.classId == id).toList();
 215:   List<AcademicStudent> classStudents(String id) =>
 216:       students.where((v) => v.belongsTo(id)).toList();
 217:   List<AcademicRecord> classProjects(String id) {
 218:     final groupIds = classGroups(id).map((g) => g.id).toSet();
 219:     return records('projects')
 220:         .where(
 221:           (p) =>
 222:               groupIds.contains(p.text('groupId')) ||
 223:               (p.text('groupId').isEmpty && p.text('classId') == id),
 224:         )
 225:         .toList();

 228:   /// Cần chỉnh sửa đã là một quyết định của lượt đánh giá, không còn ở tập đang mở.
 229:   List<AcademicRecord> classReviews(ClassRoom cls) =>
 230:       ((data['reviewState'] as AcademicJson)['requests'] as List)
 231:           .map((v) => AcademicRecord(v as AcademicJson))
 232:           .where(
 233:             (r) =>
 234:                 r.text('classCode') == cls.code &&
 235:                 ![
 236:                   'approved',
 237:                   'needsRevision',
 238:                   'notAccepted',
 239:                 ].contains(r.text('supervisorStatus')),
 240:           )
 241:           .toList();
 242:   List<AcademicLab> effectiveLabs(ClassRoom cls) =>
 243:       cls.labOverride ?? offering(cls.offeringId)?.labs ?? [];
 244:   RuleSet? effectiveRule(ClassRoom cls, [LabItem? item]) => rule(
 245:     item?.ruleSetId.isNotEmpty == true
 246:         ? item!.ruleSetId
 247:         : cls.ruleOverride.isNotEmpty
 248:         ? cls.ruleOverride
 249:         : offering(cls.offeringId)?.ruleSetId ?? '',
 250:   );
 251:   Set<String> courseRuleIds(String courseId) {
 252:     final offs = courseOfferings(courseId);
 253:     final offIds = offs.map((o) => o.id).toSet();
 254:     final cls = classes.where(
 255:       (c) => c.courseId == courseId || offIds.contains(c.offeringId),
 256:     );
 257:     return {
 258:       ...offs.map((o) => o.ruleSetId),
 259:       ...offs
 260:           .expand((o) => o.labs)
 261:           .expand((l) => l.items)
 262:           .map((i) => i.ruleSetId),
 263:       ...cls.map((c) => c.ruleOverride),
 264:       ...cls
 265:           .expand((c) => c.labOverride ?? <AcademicLab>[])
 266:           .expand((l) => l.items)
 267:           .map((i) => i.ruleSetId),
 268:     }..remove('');
 269:   }

 274:   /// Đồng bộ số liệu Dashboard từ cùng kho fake, không giữ một bản học vụ tách rời.
 275:   AcademicJson get dashboardJson {
 276:     final result = AcademicRecord(data['dashboard'] as AcademicJson).toJson();
 277:     for (final key in [
 278:       'courses',
 279:       'classes',
 280:       'semesters',
 281:       'groups',
 282:       'projects',
 283:       'registrations',
 284:     ]) {
 285:       result[key] = values(key);
 286:     }
 287:     result['academic'] = {
 288:       'courses': courses.length,
 289:       'students': students.length,
 290:     };
 291:     if (data['reviewState'] is Map) {
 292:       result['requests'] = [
 293:         for (final r in reviewRows('requests'))
 294:           (() {
 295:             final raw = evaluation(r);
 296:             final e = raw == null
 297:                 ? null
 298:                 : ReviewPolicy.currentIllustration(
 299:                     this,
 300:                     raw,
 301:                     r.text('documentTitle'),
 302:                   );
 303:             return {
 304:               ...r.toJson(),
 305:               'ai': {
 306:                 'status': raw?.text('evaluationStatus') ?? 'notRun',
 307:                 'counts': e?.text('evaluationStatus') == 'completed'
 308:                     ? {
 309:                         'errors': e!
 310:                             .values('findings')
 311:                             .where((f) => f['level'] == 'error')
 312:                             .length,
 313:                         'warnings': e
 314:                             .values('findings')
 315:                             .where((f) => f['level'] == 'warning')
 316:                             .length,
 317:                         'suggestions': e
 318:                             .values('criterionResults')
 319:                             .fold<int>(
 320:                               0,
 321:                               (n, c) => n + (c['suggestions'] as List).length,
 322:                             ),
 323:                       }
 324:                     : null,
 325:               },
 326:               'relativeTime':
 327:                   ((data['presentation'] as Map)['reviews']
 328:                       as Map)[r.id]?['relativeTime'] ??
 329:                   'Vừa xong',
 330:             };
 331:           })(),
 332:       ];
 333:     }
 334:     return result;
 335:   }
