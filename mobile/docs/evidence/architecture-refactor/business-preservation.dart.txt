import 'dart:convert';
import 'dart:io';
import 'architecture_audit.dart' show declarationTokens;

void main() {
  final map = jsonDecode(File('docs/evidence/architecture-refactor/migration-map.json').readAsStringSync()) as List;
  final selected = {
    'lib/models/session_scope.dart',
    'lib/models/academic/academic_models.dart',
    'lib/models/academic/academic_policy.dart',
    'lib/models/academic/academic_projections.dart',
    'lib/models/academic/management_queries.dart',
    'lib/models/academic/review_policy.dart',
    'lib/models/academic/project_context.dart',
    'lib/models/dashboard/dashboard_snapshot.dart',
    'lib/models/dashboard/dashboard_state.dart',
    'lib/repositories/academic/course_contract.dart',
    'lib/repositories/academic/fake_academic_repository.dart',
    'lib/services/academic/fake_academic_service.dart',
    'lib/services/academic/student_import_service.dart',
    'lib/viewmodels/academic/management_actions.dart',
    'lib/viewmodels/academic/review_actions.dart',
    'lib/viewmodels/academic/directory_view_model.dart',
  };
  final results = <String, bool>{};
  for (final row in map) {
    final old = row['before'] as String;
    if (!selected.contains(old)) continue;
    results[old] = declarationTokens(File('.local/architecture-before/$old').readAsStringSync()) == declarationTokens(File(row['after'] as String).readAsStringSync());
  }
  File('docs/evidence/architecture-refactor/business-preservation.json').writeAsStringSync(const JsonEncoder.withIndent('  ').convert(results));
  print('Critical domain/command declarations identical: ${results.values.where((v) => v).length}/${results.length}');
  if (results.values.any((v) => !v)) exitCode = 1;
}
